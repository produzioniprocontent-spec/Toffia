// Disattiva Notifiche con gestione sicura dei blocchi/timeout async function deactivateNotifications() { toggleIst.checked = false; toggleEvt.checked = false; btnDeactivateText.textContent = "Disattivazione..."; // 1. Funzione di disattivazione OneSignal con Timeout const processOneSignalDeactivation = new Promise((resolve) => { // Timeout di sicurezza: se OneSignal si blocca, proseguiamo comunque dopo 3 secondi const timer = setTimeout(() => { console.warn("OneSignal timeout: disattivazione completata in locale."); resolve(); }, 3000); window.OneSignalDeferred = window.OneSignalDeferred || []; OneSignalDeferred.push(async function(OneSignal) { try { // Aggiorniamo i Tag se l'utente è registrato if (OneSignal.User) { await OneSignal.User.addTags({ "istituzionale": "false", "eventi": "false" }); // Disattiviamo le notifiche solo se esiste una sottoscrizione attiva if (OneSignal.User.PushSubscription) { await OneSignal.User.PushSubscription.optOut(); } } } catch (e) { console.error("Errore interno SDK OneSignal:", e); } finally { clearTimeout(timer); resolve(); } }); }); // 2. Esecuzione e aggiornamento interfaccia try { await processOneSignalDeactivation; } catch (err) { console.error("Errore durante la disattivazione:", err); } finally { // Aggiorna SEMPRE il localStorage e l'interfaccia visiva localStorage.setItem('toffia_notif_ist', 'false'); localStorage.setItem('toffia_notif_evt', 'false'); localStorage.setItem('toffia_notif_activated', 'false'); localStorage.setItem('toffia_notif_deactivated', 'true'); setBtnPendingState(); setBtnDeactivatedState(); } }
